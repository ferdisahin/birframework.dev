import { Route, type Context } from 'birframework';
import { UserController } from '../app/controllers/UserController.js';

const users = new UserController();

// Basic route with parameter matching
Route.get('/users/:id', (ctx: Context) => users.show(ctx));

// Route group with validation and middleware
Route.group({ prefix: '/api/v1', middleware: ['auth'] }, () => {
  Route.post('/posts', async (ctx: Context) => {
    const data = await ctx.validate({
      title: 'required|string|min:3',
      content: 'required|string'
    });
    return ctx.json({ status: 'created', data }, 201);
  });
});

// Blade template view rendering with Instant Turbo
Route.get('/', (ctx: Context) => {
  return ctx.render('welcome', { title: 'BirFramework' });
});
import { Model, DB } from 'birframework';

export class User extends Model {
  static override table = 'users';

  // Fluent Eloquent relationships
  posts() {
    return this.hasMany('posts', 'user_id');
  }
}

// Active-record query examples
const user = await User.find(1);
const activeUsers = await User.where('status', '=', 'active')
  .orderBy('created_at', 'desc')
  .get();

// ACID transaction support across SQLite, Postgres, and MySQL
await DB.transaction(async (trx) => {
  await trx.table('accounts').where('id', 1).decrement('balance', 50);
  await trx.table('accounts').where('id', 2).increment('balance', 50);
});

Built for Full-Stack Applications

Core capabilities engineered into a single unified runtime.

Radix-Speed HTTP Routing

Parameter matching with :id and {id}, route groups with nested prefixing and middleware stacks, and native Server-Sent Events (SSE) streaming for real-time and LLM token outputs.

Instant Turbo Navigation

Server-rendered Blade templates morph into the live DOM on click. Get the fluid speed of a single-page application without the overhead of client-side React or hydration bundles.

Multi-Driver Database Layer

Active-record models, schema builders, migrations, and seeders. Uses zero-dependency native node:sqlite in local development and translates queries seamlessly to PostgreSQL or MySQL.

Queues, Scheduler & Security

In-process cron scheduling daemon, asynchronous queue worker with retry backoff, Scrypt password hashing, signed cookies, CSRF protection, and rate limiting out of the box.
