import { Route, type Context } from 'birframework';
import { UserController } from '../app/controllers/UserController.js';

const users = new UserController();

// Basic route with parameter matching
Route.get('/users/:id', (ctx: Context) => users.show(ctx));

// Route group with validation and middleware
Route.group({ prefix: '/api/v1', middleware: ['auth'] }, () => {
  Route.post('/posts', async (ctx: Context) => {
    const data = await ctx.validate({
      title: 'required|string|min:3',
      content: 'required|string'
    });
    return ctx.json({ status: 'created', data }, 201);
  });
});

// Blade template view rendering with Instant Turbo
Route.get('/', (ctx: Context) => {
  return ctx.render('welcome', { title: 'BirFramework' });
});
import { Model, DB } from 'birframework';

export class User extends Model {
  static override table = 'users';

  // Fluent Eloquent relationships
  posts() {
    return this.hasMany('posts', 'user_id');
  }
}

// Active-record query examples
const user = await User.find(1);
const activeUsers = await User.where('status', '=', 'active')
  .orderBy('created_at', 'desc')
  .get();

// ACID transaction support across SQLite, Postgres, and MySQL
await DB.transaction(async (trx) => {
  await trx.table('accounts').where('id', 1).decrement('balance', 50);
  await trx.table('accounts').where('id', 2).increment('balance', 50);
});

System Anatomy

Cohesive runtime. Zero glue code.

Rather than stitching together 8 uncoordinated libraries, BirFramework's core systems share the same context, lifecycle, and memory space.

01 / HTTP & ROUTING Radix Tree

Sub-Microsecond Dispatch

Routes are indexed in an in-memory radix tree with parameter extraction (:id, {slug}). Includes route prefixing, middleware pipelines, and native Server-Sent Events (SSE) streaming for real-time and LLM workloads.

Route.group({ prefix: '/api', middleware: ['auth'] })
02 / DATABASE & ORM Active Record

Zero-Config Local SQLite to PostgreSQL

Uses Node's native node:sqlite in local development with zero external C++ bindings. Migrations and Active Record models translate queries seamlessly to PostgreSQL or MySQL in production with ACID transactions.

await User.where('status', 'active').get()
03 / FRONTEND Blade + Turbo

Instant DOM Morphing without React Overhead

Server-rendered Blade components (<x-layouts.app>) morph seamlessly into the live DOM via BirTurbo on navigation. No heavy JavaScript hydration bundles, no layout shifts, instant page transitions.

<x-layouts.app title="Dashboard">
04 / BACKGROUND SERVICES In-Process Daemons

Queues, Scheduler & Built-in Security

In-process cron task scheduler, asynchronous queue workers with exponential retry backoff, Scrypt password hashing, signed cookies, and sliding-window rate limiting ready out of the box.
